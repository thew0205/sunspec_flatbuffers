514inline void stringToModbusRegisterFormat(const std::string &str, uint16_t *buffer, uint16_t bufferSize)
515{
516 const char *raw = str.data();
517 bufferSize = std::min(static_cast<size_t>((bufferSize * 2) - 1), str.size()); // The minus one is space for a compulsory null.
518
519 if (((bufferSize * 2) - 1) < str.size())
520 {
521 for (int i = 0; i < bufferSize; i++)
522 {
523 buffer[i] = (raw[(i * 2) + 0] << 8) | raw[(i * 2) + 1];
524 }
525 // Adding the compulsory null terminator since the buffer size is less than the string length;
526 // Remember that it is little endian which will be swap on the host side.
527 buffer[bufferSize - 1] = buffer[bufferSize] >> 8 & 0x00FF;
528 }
529 else
530 {
531 for (int i = 0; i < (str.size() + 1) / 2; i++)
532 {
533 buffer[i] = (raw[(i * 2) + 0] << 8) | raw[(i * 2) + 1];
534 }
535 // just fill the rest with null terminator.
536 // There is no need for manual swapping as the null terminator will be placed correct wither being even or odd.
537 for (int i = (str.size() + 1) / 2; i < bufferSize; i++)
538 {
539 buffer[i] = '\0';
540 }
541 }
542}
