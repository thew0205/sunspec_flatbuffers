507inline void stringToModbusRegisterFormat(const std::string &str, uint16_t *buffer, uint16_t bufferSize)
508{
509 const char *raw = str.data();
510 bufferSize = std::min(static_cast<size_t>((bufferSize * 2) - 1), str.size()); // The minus one is space for a compulsory null.
511
512 if (((bufferSize * 2) - 1) < str.size())
513 {
514 for (int i = 0; i < bufferSize; i++)
515 {
516 buffer[i] = (raw[(i * 2) + 0] << 8) | raw[(i * 2) + 1];
517 }
518 // Adding the compulsory null terminator since the buffer size is less than the string length;
519 // Remember that it is little endian which will be swap on the host side.
520 buffer[bufferSize - 1] = buffer[bufferSize] >> 8 & 0x00FF;
521 }
522 else
523 {
524 for (int i = 0; i < (str.size() + 1) / 2; i++)
525 {
526 buffer[i] = (raw[(i * 2) + 0] << 8) | raw[(i * 2) + 1];
527 }
528 // just fill the rest with null terminator.
529 // There is no need for manual swapping as the null terminator will be placed correct wither being even or odd.
530 for (int i = (str.size() + 1) / 2; i < bufferSize; i++)
531 {
532 buffer[i] = '\0';
533 }
534 }
535}
